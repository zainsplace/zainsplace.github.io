- Sealed papers total exactly 90 marks. Mocks 1 and 2 are 86 and 80 marks; leave them as they are.
- **Small change from the spec:** sealed parts are left out of the Practice summary tiles as well as Practice itself. PPP marks still go into `state.questions.history` (and so into the "latest score" for each part), and the PPP home shows its own record.

2. **A saved sitting that references a part later renamed or removed** must not break the PPP screen. Task 1 tests `pruneUnknown`, and Task 5 Step 8 edits the saved state by hand.

```js
/* ===== PPP ENGINE =====
   Pure logic for the Pure Past Papers mode on the Questions page: part clocks,
   paper rest periods, what to do next, topic sets, diagnosis checks, and
   checking saved PPP state. No DOM and no app globals, so tests/ppp.test.js
   can load it in Node. Dates are local 'YYYY-MM-DD' strings, which compare
   correctly as strings. */
const PPP = (() => {
  const REST_DAYS = 21;            // a sat paper rests this long before a re-sit
  const SECONDS_PER_MARK = 80;     // exam pace: 90 marks in 120 minutes
  const REASONS = ['knowledge', 'technique', 'misread', 'time', 'careless'];
  const TOPIC_MAX = 8;
  const MAX_SITTINGS = 100;
  const PAPERS_FOR_PHASE_3 = 2;
  const STEPS = ['answer', 'mark', 'diagnose', 'fix', 'redo', 'redomark', 'next'];

  const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;
  const ID_RE = /^[A-Za-z0-9._-]{1,64}$/;
  const UNSAFE = new Set(['__proto__', 'constructor', 'prototype']);

  const isObj = v => v !== null && typeof v === 'object' && !Array.isArray(v);
  const isStr = v => typeof v === 'string';
  const isNum = v => typeof v === 'number' && isFinite(v);
  const intIn = (v, lo, hi) => Number.isInteger(v) && v >= lo && v <= hi;
  const isDate = v => isStr(v) && DATE_RE.test(v);

  // Built from its parts so the result is local midnight everywhere.
  function addDays(date, n) {
    const [y, m, d] = date.split('-').map(Number);
    const t = new Date(y, m - 1, d + n);
    return `${t.getFullYear()}-${String(t.getMonth() + 1).padStart(2, '0')}-${String(t.getDate()).padStart(2, '0')}`;
  }

  function defaultState() {
    return { phase: 1, sittings: [], active: {} };
  }

  function partSeconds(marks) {
    return marks * SECONDS_PER_MARK;
  }

  /* ---- Sealed papers: their parts appear only in PPP, so they stay cold ---- */
  function sealedScenarios(qfile) {
    const ids = new Set();
    ((qfile && qfile.papers) || []).forEach(p => {
      if (p.sealed) (p.scenarios || []).forEach(s => ids.add(s));
    });
    return ids;
  }

  function isSealed(q, qfile) {
    return !!q.scenario && sealedScenarios(qfile).has(q.scenario);
  }

  function bankQuestions(qfile) {
    const sealed = sealedScenarios(qfile);
    return ((qfile && qfile.questions) || []).filter(q => !(q.scenario && sealed.has(q.scenario)));
  }

  function hasPPP(qfile) {
    return ((qfile && qfile.papers) || []).some(p => p.sealed);
  }

  /* ---- Paper rest ---- */
  function lastSat(sittings, paperId) {
    let last = null;
    sittings.forEach(s => {
      if (s.kind === 'paper' && s.ref === paperId && s.status === 'done' && (!last || s.date > last)) last = s.date;
    });
    return last;
  }

  function restUntil(sittings, paperId) {
    const last = lastSat(sittings, paperId);
    return last ? addDays(last, REST_DAYS) : null;
  }

  function isResting(sittings, paperId, today) {
    const until = restUntil(sittings, paperId);
    return !!until && today < until;
  }

  // A paper never sat comes first, in file order; then the one sat longest ago
  // that has finished resting. null when every paper is resting.
  function nextPaper(papers, sittings, today) {
    const fresh = papers.find(p => !lastSat(sittings, p.id));
    if (fresh) return fresh.id;
    const ready = papers
      .filter(p => !isResting(sittings, p.id, today))
      .sort((a, b) => lastSat(sittings, a.id).localeCompare(lastSat(sittings, b.id)));
    return ready.length ? ready[0].id : null;
  }

  /* ---- Phases ---- */
  function topicCounts(letters, sittings) {
    const n = {};
    letters.forEach(L => { n[L] = 0; });
    sittings.forEach(s => {
      if (s.kind === 'topic' && s.status === 'done' && s.ref in n) n[s.ref] += 1;
    });
    return n;
  }

  function nextTopicSection(letters, sittings) {
    const n = topicCounts(letters, sittings);
    return letters.reduce((best, L) => (n[L] < n[best] ? L : best), letters[0]);
  }

  function suggestPhase(phase, letters, sittings) {
    if (phase === 1) {
      const n = topicCounts(letters, sittings);
      return letters.every(L => n[L] > 0) ? 2 : null;
    }
    if (phase === 2) {
      const papers = sittings.filter(s => s.kind === 'paper' && s.status === 'done').length;
      return papers >= PAPERS_FOR_PHASE_3 ? 3 : null;
    }
    return null;
  }

  function doNext({ phase, letters, papers, sittings, today }) {
    if (phase === 1) return { kind: 'topic', section: nextTopicSection(letters, sittings) };
    const paperId = nextPaper(papers, sittings, today);
    return paperId ? { kind: 'paper', paperId, timed: phase === 3 } : { kind: 'rest' };
  }

  /* ---- Topic sets ----
     Never-tried parts first (file order), then lowest last mark. Picked across
     scenarios in turn so one scenario does not fill the set. */
  function buildTopicSet(bank, section, latest) {
    const ranked = bank.filter(q => q.section === section).sort((a, b) => {
      const ta = a.id in latest, tb = b.id in latest;
      if (ta !== tb) return ta ? 1 : -1;
      return ta ? latest[a.id] - latest[b.id] : 0;
    });
    const want = Math.min(TOPIC_MAX, ranked.length);
    const picked = [];
    const used = new Set();
    while (picked.length < want) {
      const next = ranked.find(q => !picked.includes(q) && !used.has(q.scenario || q.id));
      if (!next) { used.clear(); continue; }
      picked.push(next);
      used.add(next.scenario || next.id);
    }
    return picked.map(q => q.id);
  }

  /* ---- Diagnosis ---- */
  function lostMarks(score, marks) {
    return Math.max(0, marks - score);
  }

  // One valid reason for every lost mark.
  function partDiagnosed(reasons, lost) {
    return Array.isArray(reasons) && reasons.length === lost && reasons.every(x => REASONS.includes(x));
  }

  function needsWrittenFix(reasons) {
    return Array.isArray(reasons) && reasons.includes('knowledge');
  }

  function reasonCounts(parts) {
    const n = {};
    REASONS.forEach(r => { n[r] = 0; });
    Object.values(parts).forEach(p => (p.reasons || []).forEach(r => { if (r in n) n[r] += 1; }));
    return n;
  }

  function recordSitting(st, sitting) {
    st.sittings.push(sitting);
    if (st.sittings.length > MAX_SITTINGS) st.sittings.splice(0, st.sittings.length - MAX_SITTINGS);
  }

  /* ---- Saved state ----
     Everything here came from localStorage or an imported backup, so each
     field is checked against the shape the app writes. Anything else is
     dropped rather than trusted. */
  function cleanMap(raw, keyOk, valueOk) {
    const out = {};
    if (!isObj(raw)) return out;
    Object.keys(raw).forEach(k => {
      if (!UNSAFE.has(k) && keyOk(k) && valueOk(raw[k])) out[k] = raw[k];
    });
    return out;
  }

  const reasonList = v => Array.isArray(v) && v.every(x => REASONS.includes(x));
  const draftReasons = v => Array.isArray(v) && v.length <= 12 && v.every(x => x === null || REASONS.includes(x));
  const seconds = v => intIn(v, 0, 86400);

  function coerceSitting(x) {
    if (!isObj(x) || !isStr(x.id) || !['paper', 'topic'].includes(x.kind) || !isStr(x.ref)
        || !isDate(x.date) || !['done', 'abandoned'].includes(x.status)) return null;
    return {
      id: x.id, kind: x.kind, ref: x.ref, date: x.date, status: x.status,
      score: intIn(x.score, 0, 999) ? x.score : 0,
      max: intIn(x.max, 0, 999) ? x.max : 0,
      minutesUsed: isNum(x.minutesUsed) && x.minutesUsed >= 0 ? x.minutesUsed : 0,
      parts: cleanMap(x.parts, k => ID_RE.test(k), p => isObj(p) && intIn(p.score, 0, 99)
        && reasonList(p.reasons) && isStr(p.fix) && (p.redo === null || intIn(p.redo, 0, 99)) && seconds(p.seconds))
    };
  }

  function coerceActive(a) {
    if (!isObj(a) || !isStr(a.id) || !['paper', 'topic'].includes(a.kind) || !isStr(a.ref)
        || !STEPS.includes(a.step)) return null;
    if (!Array.isArray(a.qIds) || !a.qIds.length || !a.qIds.every(id => isStr(id) && ID_RE.test(id))) return null;
    if (!intIn(a.idx, 0, a.qIds.length - 1)) return null;
    if (!isNum(a.startedAt) || !isNum(a.partStartedAt) || !(a.partEndAt === null || isNum(a.partEndAt))) return null;
    const ids = new Set(a.qIds);
    const inSit = k => ids.has(k);
    return {
      id: a.id, kind: a.kind, ref: a.ref, timed: a.timed === true, step: a.step,
      startedAt: a.startedAt, qIds: a.qIds.slice(), idx: a.idx,
      partStartedAt: a.partStartedAt, partEndAt: a.partEndAt,
      answers: cleanMap(a.answers, k => /^ppp-[A-Za-z0-9._-]{1,80}$/.test(k), isStr),
      redoAnswers: cleanMap(a.redoAnswers, k => /^pppr-[A-Za-z0-9._-]{1,80}$/.test(k), isStr),
      scores: cleanMap(a.scores, inSit, v => intIn(v, 0, 99)),
      redoScores: cleanMap(a.redoScores, inSit, v => intIn(v, 0, 99)),
      reasons: cleanMap(a.reasons, inSit, draftReasons),
      fixes: cleanMap(a.fixes, inSit, isStr),
      seconds: cleanMap(a.seconds, inSit, seconds)
    };
  }

  function coerceState(raw) {
    const s = defaultState();
    if (!isObj(raw)) return s;
    if (intIn(raw.phase, 1, 3)) s.phase = raw.phase;
    if (Array.isArray(raw.sittings)) s.sittings = raw.sittings.map(coerceSitting).filter(Boolean).slice(-MAX_SITTINGS);
    s.active = coerceActive(raw.active) || {};
    return s;
  }

  // Drops anything that points at a part the current data no longer has.
  function pruneUnknown(st, known) {
    st.sittings.forEach(s => Object.keys(s.parts).forEach(id => { if (!known.has(id)) delete s.parts[id]; }));
    if (Array.isArray(st.active.qIds) && !st.active.qIds.every(id => known.has(id))) st.active = {};
    return st;
  }

  const api = {
    REST_DAYS, SECONDS_PER_MARK, REASONS, TOPIC_MAX, MAX_SITTINGS,
    addDays, defaultState, partSeconds, isSealed, bankQuestions, hasPPP,
    lastSat, restUntil, isResting, nextPaper,
    topicCounts, nextTopicSection, suggestPhase, doNext, buildTopicSet,
    lostMarks, partDiagnosed, needsWrittenFix, reasonCounts, recordSitting, coerceState, pruneUnknown
  };
  if (typeof module !== 'undefined' && module.exports) module.exports = api;
  return api;
})();
```

```js
       <details class="scenario-peek"><summary>Re-read the scenario</summary><div class="scenario-stem">${examText(sc.stem)}</div></details>`;
  const timed = ppp.step === 'answer' && ppp.partEndAt !== null;
  container.innerHTML = `
    <div class="mock-bar">
      <strong>${escapeHTML(pppActiveName(ppp))}</strong>
      <span class="mock-progress">Part ${ppp.idx + 1} of ${n} · ${pppScoreSoFar()} marks so far</span>
      ${timed ? '<span class="timer-display mock-timer" id="ppp-timer" role="timer" aria-label="Time left for this part">–</span>' : ''}
      <button class="btn btn-secondary btn-sm" onclick="abandonPPP()">Abandon</button>

  if (timed) updatePPPClock();
}

function pppStepHtml(q) {
  const s = ppp.scores[q.id];
  const lost = s === undefined ? 0 : PPP.lostMarks(s, q.marks);
  const remark = ppp.step === 'diagnose' || (ppp.step === 'next' && !(q.id in ppp.redoScores))
    ? `<button class="btn btn-secondary btn-sm" onclick="pppRemark()">Re-mark</button>` : '';
  switch (ppp.step) {
    case 'answer':
      return `${renderAnswerArea(q, 'ppp-' + q.id)}
        <p class="ms-note">Pure attempt: no notes. If you cannot answer it, leave it blank. The blank shows you exactly what to revise.</p>
        <button class="btn btn-primary" onclick="pppSubmitAnswer(false)">Done, mark it</button>`;
    case 'mark':
      return `<div class="mock-answer">${pppAnswerHtml(q, false)}</div>
        <p class="q-intro">Mark strictly. Give yourself no benefit of the doubt, and read the "Do not credit" list before you tick.</p>
        <div class="marker" id="marker-${q.id}">${pppMarkerHtml(q)}</div>`;
    case 'diagnose':
      return `<div class="mock-answer">${pppAnswerHtml(q, false)}</div>
        <div class="mock-marked">Marked <strong>${s}</strong>/${q.marks} ${remark}</div>
        ${pppSchemeSummary(q, false)}
        <p class="q-intro">You dropped ${pppPlural(lost, 'mark')}. Give every one a reason. Be honest: the reasons decide what you fix.</p>
        <div class="ppp-reason-key">${PPP.REASONS.map(r => `<div><strong>${PPP_REASON_LABELS[r]}:</strong> ${PPP_REASON_HINTS[r]}</div>`).join('')}</div>
        ${ppp.reasons[q.id].map((r, i) => `
          <div class="ppp-reasons" role="group" aria-label="Reason for lost mark ${i + 1}" data-reasons="${i}">
            <span class="ppp-mark-n">Mark ${i + 1}</span>
            ${PPP.REASONS.map(x => `<button class="tab-btn ${r === x ? 'active' : ''}" data-r="${x}" aria-pressed="${r === x}" onclick="pppSetReason(${i}, '${x}')">${PPP_REASON_LABELS[x]}</button>`).join('')}
          </div>`).join('')}
        <div class="ppp-actions">
          ${lost > 1 ? `<button class="btn btn-secondary btn-sm" onclick="pppSameReason()">Same reason for every mark</button>` : ''}
          <button class="btn btn-primary" onclick="pppToFix()">Next: fix it</button>
        </div>`;
    case 'fix': {
      const rs = ppp.reasons[q.id].filter((r, i, a) => a.indexOf(r) === i);
      const need = PPP.needsWrittenFix(rs);
      return `<div class="question-meta">${rs.map(r => `<span class="badge">${PPP_REASON_LABELS[r]}</span>`).join('')}</div>
        <div class="ppp-fix-links">
          <button class="btn btn-secondary btn-sm" onclick="pppOpenNotes('${q.section}')">Section ${q.section} notes</button>
          <button class="btn btn-secondary btn-sm" onclick="pppOpenCards('${q.section}')">Section ${q.section} flashcards</button>
        </div>
        ${pppSchemeSummary(q, true)}
        ${rs.filter(r => r !== 'knowledge').map(r => `<p class="ms-note"><strong>${PPP_REASON_LABELS[r]}:</strong> ${escapeHTML(pppTip(q, r))}</p>`).join('')}
        <label class="ms-title" for="pppfix-${q.id}">What the answer should have said${need ? ' (required)' : ' (optional)'}</label>
        <textarea class="quiz-answer-area" id="pppfix-${q.id}" rows="3"></textarea>
        <button class="btn btn-primary" onclick="pppToRedo()">Next: redo it from memory</button>`;
    }
    case 'redo':
      return `<p class="ms-note">Answer it again from memory. Your first answer and your correction are hidden.</p>
        ${renderAnswerArea(q, 'pppr-' + q.id)}
        <button class="btn btn-primary" onclick="pppSubmitRedo()">Done, mark the redo</button>`;
    case 'redomark':
      return `<div class="mock-answer">${pppAnswerHtml(q, true)}</div>

        <div class="marker" id="marker-${q.id}">${pppMarkerHtml(q)}</div>`;

function pppBindStep(container, q) {
  if (ppp.step === 'answer' || ppp.step === 'redo') {
    const map = ppp.step === 'answer' ? ppp.answers : ppp.redoAnswers;
    container.querySelectorAll('textarea.quiz-answer-area').forEach(t => {
      t.value = map[t.id] || '';
      t.addEventListener('input', () => { map[t.id] = t.value; persistPPP(); });
    });
  } else if (ppp.step === 'fix') {
    const t = el('pppfix-' + q.id);
    if (t) {
      t.value = ppp.fixes[q.id] || '';
      t.addEventListener('input', () => { ppp.fixes[q.id] = t.value; persistPPP(); });
    }
  } else if ((ppp.step === 'mark' || ppp.step === 'redomark') && q.markScheme) {
    updateMarker(q.id);
  }
}

function pppGo(step) {
  ppp.step = step;
  persistPPP(true);
  renderQuestions();
}

/* ---- Moving through a part ---- */
function updatePPPClock() {
  if (!ppp || ppp.step !== 'answer' || ppp.partEndAt === null) { stopPPPTimer(); return; }
  const left = Math.max(0, Math.round((ppp.partEndAt - Date.now()) / 1000));
  const t = el('ppp-timer');
  if (t) {
    t.textContent = fmtClock(left * 1000);
    t.classList.toggle('warning', left <= 60 && left > 15);
    t.classList.toggle('critical', left <= 15);
  }
  if (left === 0) {
    toast('Time is up for this part. Pens down.', 3000);
    pppSubmitAnswer(true);
  }
}

function pppSubmitAnswer(timeUp) {
  if (!ppp || ppp.step !== 'answer') return;
  stopPPPTimer();
  const q = pppQ();
  const used = Math.round((Date.now() - ppp.partStartedAt) / 1000);
  ppp.seconds[q.id] = ppp.partEndAt !== null ? Math.min(used, PPP.partSeconds(q.marks)) : Math.min(used, 86400);
  ppp.step = 'mark';
  delete markerLevel[q.id];
  persistPPP(true);
  if (currentPage === 'questions' && qMode === 'ppp') renderQuestions();
}

function recordPPPMark(qId, score) {
  if (!ppp || qId !== ppp.qIds[ppp.idx]) return;
  const q = pppQ();
  if (ppp.step === 'mark') {
    ppp.scores[qId] = score;
    const lost = PPP.lostMarks(score, q.marks);
    if (!lost) return pppGo('next');
    if (!Array.isArray(ppp.reasons[qId]) || ppp.reasons[qId].length !== lost) ppp.reasons[qId] = Array(lost).fill(null);
    pppGo('diagnose');
  } else if (ppp.step === 'redomark') {
    ppp.redoScores[qId] = score;
    pppGo('next');
  }
}

function pppRemark() {
  const q = pppQ();
  delete ppp.scores[q.id];
  delete ppp.reasons[q.id];
  delete markerLevel[q.id];
  pppGo('mark');
}

function pppSetReason(i, r) {
  const q = pppQ();
  if (!ppp || ppp.step !== 'diagnose' || !Array.isArray(ppp.reasons[q.id])) return;
  ppp.reasons[q.id][i] = r;
  persistPPP();
  const group = document.querySelector(`[data-reasons="${i}"]`);
  if (group) group.querySelectorAll('button').forEach(b => {
    const on = b.dataset.r === r;
    b.classList.toggle('active', on);
    b.setAttribute('aria-pressed', String(on));
  });
}

function pppSameReason() {
  const reasons = ppp.reasons[pppQ().id] || [];
  const r = reasons.find(Boolean);
  if (!r) { toast('Pick a reason for Mark 1 first'); return; }
  reasons.forEach((_, i) => pppSetReason(i, r));
}

function pppToFix() {
  const q = pppQ();
  if (!PPP.partDiagnosed(ppp.reasons[q.id], PPP.lostMarks(ppp.scores[q.id], q.marks))) {
    toast('Give every lost mark a reason first');
    return;
  }
  pppGo('fix');
  scrollContentTop();
}

function pppTip(q, r) {
  if (r === 'misread') return (q.traps && q.traps.length) ? q.traps.join(' ')
    : 'Underline the command word, the number asked for, and any "other than", "one" or "other" before you write.';
  if (r === 'technique') return PPP_TECHNIQUE_TIPS[q.commandWord] || PPP_TECHNIQUE_TIPS.Explain;
  if (r === 'time') return 'Use the marks as a clock: about 1 minute 20 seconds a mark. Move on and come back.';
  return 'Leave five minutes at the end to re-read every answer against its question.';
}

function pppOpenNotes(L) { navigate('sections', { section: L }); }
function pppOpenCards(L) {
  flashPracticeMode = true;
  flashFilter = L;
  navigate('flashcards');
}

function pppToRedo() {
  const q = pppQ();
  if (PPP.needsWrittenFix(ppp.reasons[q.id]) && !(ppp.fixes[q.id] || '').trim()) {
    toast("Write a correction first: you didn't know this one");
    const t = el('pppfix-' + q.id);
    if (t) t.focus();
    return;
  }
  delete markerLevel[q.id];
  pppGo('redo');
}

function pppSubmitRedo() {
  if (!ppp || ppp.step !== 'redo') return;
  delete markerLevel[pppQ().id];
  pppGo('redomark');
}

function pppNextPart() {
  if (!ppp || ppp.step !== 'next') return;
  if (ppp.idx >= ppp.qIds.length - 1) return pppFinish();
  ppp.idx += 1;
  pppStartPart();
  persistPPP(true);
  renderQuestions();
  scrollContentTop();
}

/* ---- Finishing ----
   The cold attempt is the honest measure, so it is the LAST history entry for
   each part (latestScores reads the last one). A redo straight after seeing the
   scheme is logged first, and earns no XP. */
function pppFinish() {
  const d = today();
  const st = pppState();
  const parts = {};
  const push = (id, score) => {
    const marks = findQuestion(id).marks;
    state.questions.history.push({ qId: id, marks, date: d, selfScore: Math.round(score / marks * 100) });
  };
  ppp.qIds.forEach(id => { if (id in ppp.redoScores) push(id, ppp.redoScores[id]); });
  ppp.qIds.forEach(id => {
    parts[id] = {
      score: ppp.scores[id],
      reasons: (ppp.reasons[id] || []).filter(Boolean),
      fix: ppp.fixes[id] || '',
      redo: id in ppp.redoScores ? ppp.redoScores[id] : null,
      seconds: ppp.seconds[id] || 0
    };
    push(id, ppp.scores[id]);
  });
  const score = ppp.qIds.reduce((n, id) => n + ppp.scores[id], 0);
  const max = pppMarksOf(ppp.qIds);
  const minutesUsed = Math.round(Object.values(ppp.seconds).reduce((a, b) => a + b, 0) / 60);
  PPP.recordSitting(st, { id: ppp.id, kind: ppp.kind, ref: ppp.ref, date: d, status: 'done',
                          score, max, minutesUsed, parts });
  state.xp = (state.xp || 0) + score * 2;
  ppp.result = { score, max, parts, minutesUsed, xp: score * 2 };
  ppp.step = 'done';
  persistPPP(true);
  bumpActivity();
  renderQuestions();
  scrollContentTop();
}

function renderPPPDone(container) {
  const r = ppp.result;
  const pct = r.max ? Math.round(r.score / r.max * 100) : 0;
  const counts = PPP.reasonCounts(r.parts);
  const lostTotal = Object.values(counts).reduce((a, b) => a + b, 0);
  const row = (label, g, m) => `
    <div class="mock-row"><span>${escapeHTML(label)}</span>
      <div class="bar"><div class="bar-fill" style="width:${m ? Math.round(g / m * 100) : 0}%"></div></div>
      <strong>${g}/${m}</strong></div>`;
  const paper = ppp.kind === 'paper' ? mockPaper(ppp.ref) : null;
  const rows = paper
    ? paper.scenarios.map((sid, i) => {
        const ids = ppp.qIds.filter(id => findQuestion(id).scenario === sid);
        return row(`Q${i + 1} ${qData.scenarios.find(s => s.id === sid).title}`,
                   ids.reduce((n, id) => n + r.parts[id].score, 0), pppMarksOf(ids));
      }).join('')
    : ppp.qIds.map(id => row(pppPartLabel(findQuestion(id)), r.parts[id].score, findQuestion(id).marks)).join('');
  container.innerHTML = `
    <div class="card mock-result">
      <div class="mock-paper-meta">${escapeHTML(pppActiveName(ppp))} · ${r.minutesUsed} minutes answering</div>
      <div class="mock-score">${r.score}<span>/${r.max}</span></div>
      <div class="mock-paper-meta">${pct}% · +${r.xp} XP</div>
    </div>
    <h3 style="margin:18px 0 8px">${paper ? 'By question' : 'By part'}</h3>
    ${rows}
    <h3 style="margin:18px 0 8px">Where the marks went</h3>
    ${lostTotal ? `<div class="ppp-reason-counts">${PPP.REASONS.filter(k => counts[k]).map(k => `
      <div><strong>${counts[k]}</strong><span>${PPP_REASON_LABELS[k]}</span></div>`).join('')}</div>`
      : '<p class="q-intro">Full marks. Nothing lost.</p>'}
    ${paper ? `<p class="ms-note" style="margin-top:12px">This paper now rests for ${PPP.REST_DAYS} days, so you forget the answers before you sit it again.</p>` : ''}
    <div style="display:flex;gap:8px;flex-wrap:wrap;margin-top:12px">
      <button class="btn btn-primary btn-sm" onclick="pppHome()">Back to PPP</button>
    </div>`;
}
```
